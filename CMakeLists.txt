# Build system index. Rationale: docs/BUILD-SYSTEM.md (bodies there, 1-line pointers here).
# The root is libraries, install, toolchains, frontend discovery, then test includes (issue #271).
# clangd needs a compilation database (gitignored).
cmake_minimum_required(VERSION 3.22)
project(hunmin LANGUAGES C VERSION 0.3.0)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# MSVC exports via CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS; PE flags stay MinGW-only (BUILD-SYSTEM).
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
# Every .c/.h is UTF-8; /utf-8 keeps MSVC literals byte-correct (BUILD-SYSTEM).
if(MSVC)
  add_compile_options(/std:c11 /W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra)
endif()
option(BUILD_TESTING "" ON)
include(CTest)
include(FetchContent)
find_package(PkgConfig QUIET)
find_package(Python3 QUIET COMPONENTS Interpreter)
# Packaged TIP shape: static CRT so a clean image needs no VC redist (BUILD-SYSTEM).
option(HUNMIN_MSVC_STATIC_RUNTIME "Link the MSVC CRT statically (/MT); the packaged TIP shape (issue #99)" OFF)
if(HUNMIN_MSVC_STATIC_RUNTIME AND MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
# One source list, shared for POSIX and static for the Windows TSF TIP (BUILD-SYSTEM).
set(HUNMIN_CORE_SOURCES libhunmin/src/version.c libhunmin/src/tables.c libhunmin/src/schema_loader.c libhunmin/src/expr.c libhunmin/src/ic.c libhunmin/src/buffered_preedit.c libhunmin/src/candidate_tables.c libhunmin/src/wordset.c libhunmin/src/hangul_valid.c third_party/cJSON/cJSON.c)
add_library(hunmin SHARED ${HUNMIN_CORE_SOURCES})
target_include_directories(hunmin PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/libhunmin/include)
target_include_directories(hunmin PRIVATE third_party/cJSON libhunmin/src)
set_target_properties(hunmin PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(hunmin PRIVATE HUNMIN_BUILDING_DLL HUNMIN_DEFAULT_DATA_DIR=\"${CMAKE_INSTALL_FULL_DATADIR}/hunmin\")
if(WIN32 AND NOT MSVC)
  target_link_options(hunmin PRIVATE -Wl,--export-all-symbols)
endif()
add_library(hunmin_static STATIC ${HUNMIN_CORE_SOURCES})
target_include_directories(hunmin_static PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/libhunmin/include)
target_include_directories(hunmin_static PRIVATE third_party/cJSON libhunmin/src)
set_target_properties(hunmin_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(hunmin_static PUBLIC HUNMIN_STATIC)
target_compile_definitions(hunmin_static PRIVATE HUNMIN_DEFAULT_DATA_DIR=\"${CMAKE_INSTALL_FULL_DATADIR}/hunmin\")
install(FILES libhunmin/include/hunmin.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
# data, in two parts (see docs/BUILD-SYSTEM.md#data-installs-as-two-components).
install(DIRECTORY libhunmin/data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/hunmin COMPONENT data PATTERN "modes" EXCLUDE)
install(DIRECTORY libhunmin/data/modes/ DESTINATION ${CMAKE_INSTALL_DATADIR}/hunmin/modes COMPONENT modes-data)
set(HUNMIN_HAS_LIBHANGUL OFF)
if(PkgConfig_FOUND)
  pkg_check_modules(LIBHANGUL QUIET IMPORTED_TARGET GLOBAL libhangul)
  if(LIBHANGUL_FOUND)
    add_executable(gen-golden-libhangul tools/gen-golden-libhangul.c)
    target_link_libraries(gen-golden-libhangul PRIVATE PkgConfig::LIBHANGUL)
    set(HUNMIN_HAS_LIBHANGUL ON)
  endif()
endif()
# One shell for tests needing POSIX sh; Windows has no sh on PATH (BUILD-SYSTEM).
find_program(HUNMIN_SH sh PATHS /bin /usr/bin HINTS "$ENV{ProgramFiles}/Git/usr/bin" "$ENV{ProgramFiles}/Git/bin")
if(HUNMIN_HAS_LIBHANGUL AND HUNMIN_SH)
  add_custom_target(goldens COMMAND ${HUNMIN_SH} ${CMAKE_CURRENT_SOURCE_DIR}/tools/regen-goldens.sh ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
endif()
# hunmin-sim is plain C; tier-A goldens run through it, also under MSVC (issue #85).
add_executable(hunmin-sim tools/sim.c)
target_include_directories(hunmin-sim PRIVATE third_party/cJSON)
target_link_libraries(hunmin-sim PRIVATE hunmin)
if(NOT MSVC)
  install(TARGETS hunmin hunmin-sim LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
else()
  install(TARGETS hunmin LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
# GoogleTest via FetchContent; INSTALL_GTEST OFF keeps the SDK out of the packages (BUILD-SYSTEM).
if(BUILD_TESTING AND Python3_FOUND)
  enable_language(CXX)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.17.0 GIT_SHALLOW TRUE)
  option(INSTALL_GTEST "Install the fetched GoogleTest SDK alongside hunmin" OFF)
  FetchContent_MakeAvailable(googletest)
  if(MINGW)
    set(HUNMIN_TEST_LINK_OPTS -static)
  endif()
endif()
# fcitx5 floor: Fcitx5Core >= 5.1.13 (BUILD-SYSTEM); skipped when unavailable.
option(HUNMIN_FCITX5 "Build the fcitx5 addon" ON)
if(HUNMIN_FCITX5)
  find_package(Fcitx5Core 5.1.13 QUIET)
  find_package(Fcitx5Module QUIET COMPONENTS TestFrontend)
  if(Fcitx5Core_FOUND)
    add_subdirectory(frontends/fcitx5)
    set(HUNMIN_HAS_FCITX5 TRUE)
  else()
    message(STATUS "Fcitx5Core not found: fcitx5 addon disabled")
  endif()
endif()
# Windows TIP is MSVC-only; POSIX configures add nothing.
if(WIN32 AND MSVC)
  add_subdirectory(frontends/tsf)
endif()
# Tests: one file per coverage subject; bodies in cmake/Hunmin*.cmake (BUILD-SYSTEM file-layout).
if(BUILD_TESTING)
  include(cmake/HunminTestHelpers.cmake)
  include(cmake/HunminLibhunminTests.cmake)
  include(cmake/HunminEngineTests.cmake)
  include(cmake/HunminDataTests.cmake)
  include(cmake/HunminConformanceTests.cmake)
  include(cmake/HunminFrontendTests.cmake)
  include(cmake/HunminAndroidTests.cmake)
  include(cmake/HunminProcessTests.cmake)
  get_property(HUNMIN_TESTS DIRECTORY PROPERTY TESTS)
  if(HUNMIN_TESTS)
    set_tests_properties(${HUNMIN_TESTS} PROPERTIES ENVIRONMENT_MODIFICATION "PYTHONUTF8=set:1")
  endif()
endif()
